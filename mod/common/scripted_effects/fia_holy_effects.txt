fia_holy_global_tick = {
    s:STATE_PALESTINE = {
        ordered_scope_state = {
            order_by = state_population
            max = 1
            save_scope_as = fia_holy_sample
        }
    }
    if = {
        limit = { exists = scope:fia_holy_sample }
        if = {
            limit = { NOT = { has_global_variable = fia_holy_epoch } }
            set_global_variable = { name = fia_holy_epoch value = 0 }
            set_global_variable = { name = fia_holy_severity value = 0 }
            set_global_variable = { name = fia_holy_owner value = scope:fia_holy_sample.owner }
        }
        set_global_variable = { name = fia_holy_state value = scope:fia_holy_sample }
        set_global_variable = { name = fia_holy_previous_severity value = global_var:fia_holy_severity }
        if = {
            limit = { NOT = { global_var:fia_holy_owner = scope:fia_holy_sample.owner } }
            scope:fia_holy_sample = { remove_variable = fia_holy_restricted }
            set_global_variable = { name = fia_holy_owner value = scope:fia_holy_sample.owner }
            set_global_variable = { name = fia_holy_transition value = 1 days = 365 }
            change_global_variable = { name = fia_holy_epoch add = 1 }
        }
        set_global_variable = { name = fia_holy_severity value = 0 }
        if = { limit = { has_global_variable = fia_holy_transition } set_global_variable = { name = fia_holy_severity value = 1 } }
        scope:fia_holy_sample = {
            if = {
                limit = { OR = { has_variable = fia_holy_restricted AND = { devastation >= 0.25 owner = { is_at_war = yes } } } }
                set_global_variable = { name = fia_holy_severity value = 2 }
            }
            if = { limit = { devastation >= 0.5 } set_global_variable = { name = fia_holy_severity value = 3 } }
        }
        if = {
            limit = { global_var:fia_holy_severity > global_var:fia_holy_previous_severity }
            change_global_variable = { name = fia_holy_epoch add = 1 }
        }
    }
    else = {
        remove_global_variable = fia_holy_state
        set_global_variable = { name = fia_holy_severity value = 0 }
    }
}
fia_holy_country_tick = {
    if = {
        limit = { has_global_variable = fia_holy_epoch exists = global_var:fia_holy_state }
        set_variable = { name = fia_holy_severity_display value = global_var:fia_holy_severity }
        if = { limit = { NOT = { has_journal_entry = je_fia_holy } } add_journal_entry = { type = je_fia_holy } }
            fia_holy_catholic_tick = yes
    fia_holy_protestant_tick = yes
    fia_holy_orthodox_tick = yes
    fia_holy_sunni_tick = yes
    fia_holy_shiite_tick = yes
    fia_holy_jewish_tick = yes

    else = {
            fia_holy_catholic_clear = yes
    fia_holy_protestant_clear = yes
    fia_holy_orthodox_clear = yes
    fia_holy_sunni_clear = yes
    fia_holy_shiite_clear = yes
    fia_holy_jewish_clear = yes
        set_variable = { name = fia_holy_severity_display value = 0 }
    }
}

fia_holy_catholic_clear = {
    remove_variable = fia_holy_catholic_age
    remove_variable = fia_holy_catholic_response
    remove_variable = fia_holy_catholic_severity
    remove_variable = fia_holy_catholic_importance
}
fia_holy_catholic_success = {
    if = {
        limit = { NOT = { has_variable = fia_holy_catholic_reward_cap } }
        add_loyalists = { religion = rel:catholic value = fia_holy_reward }
        set_variable = { name = fia_holy_catholic_reward_cap value = 1 days = fia_holy_reward_cap_days }
        change_variable = { name = fia_holy_catholic_influence add = 5 }
        change_variable = { name = fia_holy_catholic_authority add = 5 }
        add_modifier = { name = fia_holy_credibility months = 24 }
        if = { limit = { religion = rel:catholic } add_modifier = { name = fia_holy_domestic_trust months = 24 } }
    }
}
fia_holy_catholic_failure = {
    if = {
        limit = { NOT = { has_variable = fia_holy_catholic_penalty_cap } }
        if = {
            limit = { var:fia_holy_catholic_response = 3 }
            add_radicals = { religion = rel:catholic value = fia_holy_broken_promise_rate }
            change_variable = { name = fia_holy_catholic_influence subtract = 10 }
            change_variable = { name = fia_holy_catholic_authority subtract = 10 }
            if = { limit = { religion = rel:catholic } add_modifier = { name = fia_holy_broken_promise months = 12 } }
        }
        else_if = {
            limit = { has_global_variable = fia_holy_catholic_restraint }
            add_radicals = { religion = rel:catholic value = fia_holy_neglect_low }
        }
        else_if = {
            limit = { OR = { var:fia_holy_catholic_severity >= 2 has_global_variable = fia_holy_catholic_call } }
            if = {
                limit = { religion = rel:catholic fia_restrictive_policy = yes }
                add_radicals = { religion = rel:catholic value = fia_holy_national_mobilization }
            }
            else_if = {
                limit = { var:fia_holy_catholic_importance >= 85 }
                add_radicals = { religion = rel:catholic value = fia_holy_neglect_major }
            }
            else = { add_radicals = { religion = rel:catholic value = fia_holy_neglect_minor } }
            change_variable = { name = fia_holy_catholic_influence subtract = 5 }
        }
        else = { add_radicals = { religion = rel:catholic value = fia_holy_neglect_low } }
        set_variable = { name = fia_holy_catholic_penalty_cap value = 1 days = fia_holy_reward_cap_days }
    }
}
fia_holy_catholic_tick = {
    if = {
        limit = { NOT = { has_variable = fia_holy_catholic_influence } }
        set_variable = { name = fia_holy_catholic_influence value = 10 }
        set_variable = { name = fia_holy_catholic_authority value = 30 }
        set_variable = { name = fia_holy_catholic_seen value = 0 }
        if = { limit = { this = c:PAP religion = rel:catholic } set_variable = { name = fia_holy_catholic_authority value = 60 } }
    }
    if = { limit = { is_subject = yes } change_variable = { name = fia_holy_catholic_authority subtract = 1 } }
    if = {
        limit = { fia_holy_catholic_affected = yes global_var:fia_holy_epoch > var:fia_holy_catholic_seen }
        # New epochs supersede stale petitions, never score the old one as failure.
        fia_holy_catholic_clear = yes
        set_variable = { name = fia_holy_catholic_seen value = global_var:fia_holy_epoch }
        if = {
            limit = { global_var:fia_holy_severity >= 1 }
            set_variable = { name = fia_holy_catholic_epoch value = global_var:fia_holy_epoch }
            set_variable = { name = fia_holy_catholic_age value = 0 }
            set_variable = { name = fia_holy_catholic_response value = 0 }
            set_variable = { name = fia_holy_catholic_severity value = global_var:fia_holy_severity }
            set_variable = { name = fia_holy_catholic_importance value = 80 }
            trigger_event = { id = fia.100 popup = yes }
        }
    }
    if = {
        limit = { has_variable = fia_holy_catholic_age }
        if = {
            limit = { fia_holy_catholic_pending = no }
            fia_holy_catholic_clear = yes
        }
        else = {
            change_variable = { name = fia_holy_catholic_age add = 1 }
            if = {
                limit = { fia_holy_catholic_affected = no }
                fia_holy_catholic_clear = yes
            }
            else_if = {
                limit = { fia_holy_is_resolved = yes var:fia_holy_catholic_age >= fia_holy_settlement_delay }
                if = { limit = { var:fia_holy_catholic_response >= 1 } fia_holy_catholic_success = yes }
                fia_holy_catholic_clear = yes
            }
            else_if = {
                limit = {
                    var:fia_holy_catholic_age >= fia_holy_response_months
                    OR = {
                        var:fia_holy_catholic_age >= fia_holy_war_deadline
                        NOT = { has_war_with = global_var:fia_holy_owner }
                    }
                }
                if = {
                    limit = { OR = { var:fia_holy_catholic_response = 0 var:fia_holy_catholic_response = 3 } }
                    fia_holy_catholic_failure = yes
                }
                fia_holy_catholic_clear = yes
            }
        }
    }
    clamp_variable = { name = fia_holy_catholic_influence min = 0 max = 100 }
    clamp_variable = { name = fia_holy_catholic_authority min = 0 max = 100 }
}
