fia_holy_is_custodian = {
    exists = global_var:fia_holy_state
    global_var:fia_holy_state = { owner = prev }
}
fia_holy_is_resolved = {
    exists = global_var:fia_holy_state
    global_var:fia_holy_severity <= 1
    global_var:fia_holy_state = { NOT = { has_variable = fia_holy_restricted } }
}
fia_holy_catholic_affected = {
    religion_percent_country = { target = rel:catholic value >= fia_holy_minimum_share }
    any_scope_pop = { has_pop_religion = rel:catholic total_size >= fia_minimum_population }
}
fia_holy_catholic_voice = {
    religion = rel:catholic
    is_subject = no
    OR = { has_law = law_type:law_theocracy fia_restrictive_policy = yes }
    prestige >= 50
    any_interest_group = { is_interest_group_type = ig_devout ig_clout >= fia_holy_devout_clout }
}
fia_holy_catholic_pending = {
    has_variable = fia_holy_catholic_age
    has_variable = fia_holy_catholic_epoch
    var:fia_holy_catholic_epoch = global_var:fia_holy_epoch
}
