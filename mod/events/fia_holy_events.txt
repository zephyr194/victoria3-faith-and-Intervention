fia.100 = {
    type = country_event
    placement = root
    title = fia.100.t
    desc = fia.100.d
    duration = 30
    trigger = { fia_holy_catholic_pending = yes }
    immediate = { save_scope_value_as = { name = fia_holy_event_epoch value = var:fia_holy_catholic_epoch } }
    cancellation_trigger = { OR = { fia_holy_catholic_pending = no NOT = { scope:fia_holy_event_epoch = var:fia_holy_catholic_epoch } } }
    option = {
        name = fia.100.ignore
        default_option = yes
        trigger = { fia_holy_catholic_pending = yes scope:fia_holy_event_epoch = var:fia_holy_catholic_epoch }
        set_variable = { name = fia_holy_catholic_response value = 0 }

        ai_chance = { base = 30 }
    }
    option = {
        name = fia.100.protest

        trigger = { fia_holy_catholic_pending = yes scope:fia_holy_event_epoch = var:fia_holy_catholic_epoch }
        set_variable = { name = fia_holy_catholic_response value = 1 }
        add_modifier = { name = fia_holy_relief_cost months = 6 }
        ai_chance = { base = 35 }
    }
    option = {
        name = fia.100.relief

        trigger = { fia_holy_catholic_pending = yes scope:fia_holy_event_epoch = var:fia_holy_catholic_epoch }
        set_variable = { name = fia_holy_catholic_response value = 2 }
        add_modifier = { name = fia_holy_relief_cost months = 6 } add_modifier = { name = fia_holy_relief_funding months = 6 }
        ai_chance = { base = 25 }
    }
    option = {
        name = fia.100.pledge

        trigger = { fia_holy_catholic_pending = yes scope:fia_holy_event_epoch = var:fia_holy_catholic_epoch }
        set_variable = { name = fia_holy_catholic_response value = 3 }
