namespace = fia

fia.1 = {
    type = country_event
    placement = root
    title = fia.1.t
    desc = fia.1.d
    flavor = fia.1.f
    duration = 30
    trigger = {
        exists = var:fia_domestic_state
        var:fia_domestic_state = { owner = root has_variable = fia_tension NOT = { has_variable = fia_accord_months } }
    }
    immediate = { var:fia_domestic_state = { save_scope_as = fia_event_state } }
    cancellation_trigger = {
        OR = {
            NOT = { exists = scope:fia_event_state }
            NOT = { scope:fia_event_state = { owner = root has_variable = fia_tension NOT = { has_variable = fia_accord_months } } }
        }
    }
    option = {
        name = fia.1.a
        default_option = yes
        scope:fia_event_state = { fia_conciliate = yes }
        ai_chance = { base = 40 }
    }
    option = {
        name = fia.1.b
        scope:fia_event_state = {
            change_variable = { name = fia_tension subtract = 5 }
            clamp_variable = { name = fia_tension min = 0 max = fia_maximum_tension }
        }
        add_modifier = { name = fia_mediation_cost months = 6 }
        ai_chance = { base = 35 }
    }
    option = {
        name = fia.1.c
        scope:fia_event_state = { fia_repress = yes }
        ai_chance = { base = 25 }
    }
}
